// =============================== // آدرس شیت را بعداً اینجا قرار بده // =============================== const SHEET_URL = "https://docs.google.com/spreadsheets/d/1OImD12FCHFiDxzN-h8lBWox2gM2jCmaAntPI_NjfWsg/edit?usp=drivesdk"; // لیست کالاها let products = []; // دریافت اطلاعات async function loadProducts() { if (!SHEET_URL) { document.getElementById("loading").style.display = "none"; document.getElementById("error").innerHTML = "لینک Google Sheet تنظیم نشده است."; return; } try { const res = await fetch(SHEET_URL); const text = await res.text(); const json = JSON.parse( text.substring(47).slice(0, -2) ); const rows = json.table.rows; products = []; rows.forEach(row => { if (!row.c) return; products.push({ code: row.c[0]?.v ?? "", name: row.c[1]?.v ?? "", price: row.c[2]?.v ?? "" }); }); showProducts(products); document.getElementById("loading").style.display = "none"; document.getElementById("table").style.display = "table"; } catch (e) { document.getElementById("loading").style.display = "none"; document.getElementById("error").innerHTML = "خطا در دریافت اطلاعات از شیت."; console.log(e); } } // نمایش کالاها function showProducts(list) { const tbody = document.getElementById("tbody"); tbody.innerHTML = ""; list.forEach(item => { tbody.innerHTML += ` ${item.code} ${item.name} ${Number(item.price).toLocaleString("fa-IR")} تومان `; }); } // جستجو document.getElementById("search").addEventListener("input", function () { const value = this.value.trim(); const result = products.filter(item => String(item.code).includes(value) || String(item.name).includes(value) ); showProducts(result); }); // شروع برنامه loadProducts();
